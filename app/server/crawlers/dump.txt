===================================
projects table
===================================

Table name: projects
fields:
		'project_id INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT',
		'project_name TEXT',
		'data_source TEXT',
		'jenkins_url TEXT',
		'jenkins_user TEXT',
		'jenkins_password TEXT',
		'status TEXT',
		'created TIMESTAMP',
		'last_modified TIMESTAMP'
